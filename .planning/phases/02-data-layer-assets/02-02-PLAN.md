```typescript
// types.ts
export type WeaveType = 'Plano' | 'Punto';

export type PrintRoute =
  | 'Unicolor'
  | 'Rotativa'
  | 'Davos'
  | 'Sublimacion';

export interface Technology {
  readonly id: string;
  readonly name: string;
  readonly icon: string;       // Path relativo a /public, '' si no tiene logo
  readonly description: string;
}

export interface Fabric {
  readonly id: string;          // kebab-case slug generado del nombre
  readonly name: string;        // Nombre EXACTO del PDF, sin modificar
  readonly base: string;        // Codigo BASE unico del PDF (ej: "23243")
  readonly composition: string; // Composicion textual tal cual del PDF
  readonly weave: WeaveType;
  readonly weight: string;      // Con tolerancia: "110 +-10 g/m2"
  readonly width: string;       // Con tolerancia: "151 +- 2 cm"
  readonly technologies: readonly string[];     // IDs de tecnologias aplicables
  readonly printRoutes: readonly PrintRoute[];  // Rutas de estampacion compatibles
  readonly image: string;       // Path relativo a /public/images/products/
  readonly isNew?: boolean;     // Solo Apolo y Celta tienen flag "NUEVA"
}

export interface Category {
  readonly id: string;              // kebab-case slug
  readonly name: string;            // Nombre EXACTO del PDF
  readonly color: string;           // Hex color sincronizado con globals.css @theme
  readonly foregroundColor: string;  // Color texto WCAG AA (blanco o #1a1a1a)
  readonly fabricIds: readonly string[];  // IDs de telas en esta categoria
  readonly description?: string;    // Subtitulo entre parentesis del PDF, si aplica
}
```
